<script>
  (function() {
    // ---------- MBTI 题目数据 ----------
    const questions = [
      { id: 1, text: "在聚会中，你通常更倾向于：", dimension: "EI", options: ["与很多人交流，结识新朋友", "与少数熟人深入交谈"] },
      { id: 2, text: "当你需要充电恢复精力时，你会：", dimension: "EI", options: ["参加社交活动或找人出去玩", "独自安静地看书或冥想"] },
      { id: 3, text: "你更欣赏哪种描述？", dimension: "SN", options: ["富有想象力和可能性", "注重事实和实际细节"] },
      { id: 4, text: "学习新事物时，你更喜欢：", dimension: "SN", options: ["先理解整体概念和未来应用", "按步骤学习具体事实和细节"] },
      { id: 5, text: "做重要决定时，你更依赖：", dimension: "TF", options: ["个人价值观和他人感受", "逻辑分析和客观标准"] },
      { id: 6, text: "当朋友向你倾诉烦恼，你通常：", dimension: "TF", options: ["先共情并表达支持", "先分析问题并提供解决方案"] },
      { id: 7, text: "对待日程和计划，你倾向于：", dimension: "JP", options: ["灵活开放，享受惊喜", "有条理，喜欢提前安排"] },
      { id: 8, text: "面对一项任务，你的风格接近：", dimension: "JP", options: ["随性开始，边做边调整", "制定详细计划再逐步执行"] },
      { id: 9, text: "哪种描述更符合你？", dimension: "EI", options: ["外向活泼，行动后思考", "内敛沉静，思考后行动"] },
      { id: 10, text: "阅读时，你更关注：", dimension: "SN", options: ["字里行间的隐喻和可能性", "清晰呈现的事实与信息"] },
      { id: 11, text: "批评别人时，你更可能：", dimension: "TF", options: ["注意不伤害对方感情", "直截了当指出逻辑问题"] },
      { id: 12, text: "你的工作空间通常：", dimension: "JP", options: ["有些随意但充满灵感", "整洁有序、分类清晰"] }
    ];

    // 维度映射: EI, SN, TF, JP
    const dimensionMap = {
      "EI": { left: "E", right: "I" },
      "SN": { left: "S", right: "N" },
      "TF": { left: "T", right: "F" },
      "JP": { left: "J", right: "P" }
    };

    // MBTI 类型描述库 (简洁版)
    const typeDescriptions = {
      "ISTJ": "安静、严肃，通过全面性和可靠性获得成功。实际，有责任感。",
      "ISFJ": "安静、友好、有责任心和良知。坚定地致力于完成他们的义务。",
      "INFJ": "寻求思想、关系、物质等之间的意义和联系。希望了解什么能够激励人。",
      "INTJ": "有独创的思想和强大的动力来实现自己的想法。迅速看到规律与前景。",
      "ISTP": "灵活、安静的观察者，遇到问题会迅速行动以找到可行的解决方案。",
      "ISFP": "安静、友善、敏感、温和。享受当下和周围发生的事。",
      "INFP": "理想主义，忠诚于自己的价值观和重要的人。希望外部生活与内心价值观统一。",
      "INTP": "寻求合乎逻辑的解释，对理论感兴趣。安静、有保留、灵活。",
      "ESTP": "灵活、容忍，采取务实方法取得立即结果。享受与人相处的活跃时刻。",
      "ESFP": "外向、友善、接受性强。热爱生活、人群和物质享受。",
      "ENFP": "热情洋溢、富有想象力。认为人生充满可能性。能迅速将信息联系起来。",
      "ENTP": "反应快、睿智，善于激励别人。乐于解决新挑战。",
      "ESTJ": "实际、果断，迅速采取行动。注重效率与日常细节。",
      "ESFJ": "热心、有责任心、合作。希望得到认可并营造和谐氛围。",
      "ENFJ": "热情、有同情心、反应迅速。关注他人的成长与情感。",
      "ENTJ": "坦率、果断，乐于承担领导角色。善于长远规划与目标设定。"
    };

    // 应用状态
    const state = {
      currentQuestionIndex: 0,        // 当前题目索引 0~11
      answers: new Array(questions.length).fill(null), // 存储每道题答案: 'left' 或 'right'
      testCompleted: false,
      mbtiResult: null
    };

    // 获取根容器
    const appContainer = document.getElementById('appContainer');

    // ---------- 工具函数 ----------
    function calculateMBTI() {
      const scores = {
        E: 0, I: 0,
        S: 0, N: 0,
        T: 0, F: 0,
        J: 0, P: 0
      };

      questions.forEach((q, index) => {
        const answer = state.answers[index];
        if (!answer) return; // 理论上不会发生，因为完成时全部答完
        
        const dim = q.dimension;
        const mapping = dimensionMap[dim];
        if (answer === 'left') {
          // 左侧选项对应 left 维度 (例如 E, S, T, J)
          scores[mapping.left] += 1;
        } else if (answer === 'right') {
          scores[mapping.right] += 1;
        }
      });

      // 构建MBTI类型
      const type = 
        (scores.E >= scores.I ? 'E' : 'I') +
        (scores.S >= scores.N ? 'S' : 'N') +
        (scores.T >= scores.F ? 'T' : 'F') +
        (scores.J >= scores.P ? 'J' : 'P');
      
      return type;
    }

    // 处理选项点击
    function selectOption(questionIndex, optionValue) {
      if (state.testCompleted) return;
      if (state.answers[questionIndex] === optionValue) {
        // 如果点击同一个选项：取消选中（允许修改）
        state.answers[questionIndex] = null;
      } else {
        state.answers[questionIndex] = optionValue;
      }
      renderApp();
    }

    // 导航到下一题
    function goToNext() {
      if (state.currentQuestionIndex < questions.length - 1) {
        state.currentQuestionIndex++;
        renderApp();
      }
    }

    // 返回上一题
    function goToPrev() {
      if (state.currentQuestionIndex > 0) {
        state.currentQuestionIndex--;
        renderApp();
      }
    }

    // 完成测试
    function finishTest() {
      // 检查是否所有题目都已作答
      const allAnswered = state.answers.every(ans => ans !== null);
      if (!allAnswered) {
        alert('请回答完所有题目后再查看结果。');
        return;
      }
      
      const resultType = calculateMBTI();
      state.mbtiResult = resultType;
      state.testCompleted = true;
      renderApp();
    }

    // 重新开始测试
    function restartTest() {
      state.currentQuestionIndex = 0;
      state.answers = new Array(questions.length).fill(null);
      state.testCompleted = false;
      state.mbtiResult = null;
      renderApp();
    }

    // ---------- 渲染函数 ----------
    function renderApp() {
      if (state.testCompleted && state.mbtiResult) {
        renderResultScreen();
        return;
      }

      renderTestScreen();
    }

    // 测试界面
    function renderTestScreen() {
      const qIndex = state.currentQuestionIndex;
      const question = questions[qIndex];
      const selectedAnswer = state.answers[qIndex];
      const totalQuestions = questions.length;
      const answeredCount = state.answers.filter(a => a !== null).length;
      const progressPercent = Math.round((answeredCount / totalQuestions) * 100);

      const isFirst = qIndex === 0;
      const isLast = qIndex === totalQuestions - 1;
      const currentAnswerSelected = selectedAnswer !== null;

      // 构建选项HTML
      let optionsHtml = '';
      question.options.forEach((optText, idx) => {
        const optionValue = idx === 0 ? 'left' : 'right';
        const selectedClass = (selectedAnswer === optionValue) ? 'selected' : '';
        optionsHtml += `
          <div class="option-btn ${selectedClass}" data-option-value="${optionValue}">
            <span>${optText}</span>
            <span style="font-size:1.3rem; opacity:0.7;">${idx === 0 ? '🔹' : '🔸'}</span>
          </div>
        `;
      });

      const html = `
        <h1>✨ MBTI 人格测试</h1>
        <div class="subhead">发现真实的自己</div>
        
        <div class="progress-section">
          <div class="progress-bar-bg">
            <div class="progress-fill" style="width: ${progressPercent}%;"></div>
          </div>
          <div class="question-counter">
            <span>问题 ${qIndex+1}/${totalQuestions}</span>
            <span>${answeredCount} 题已答</span>
          </div>
        </div>

        <div class="test-card">
          <div class="question-text">${question.text}</div>
          <div class="options-grid" id="optionsContainer">
            ${optionsHtml}
          </div>
        </div>

        <div class="nav-buttons">
          <button class="btn" id="prevBtn" ${isFirst ? 'disabled' : ''}>← 上一题</button>
          ${isLast ? 
            `<button class="btn primary" id="finishBtn">完成测试 ✦</button>` : 
            `<button class="btn primary" id="nextBtn" ${!currentAnswerSelected ? 'disabled' : ''}>下一题 →</button>`
          }
        </div>
        <div style="text-align:center; margin-top:1rem; color:#aa8e7a; font-size:0.8rem;">
          点击选项选中，可修改
        </div>
      `;

      appContainer.innerHTML = html;

      // 绑定事件
      const optionsContainer = document.getElementById('optionsContainer');
      if (optionsContainer) {
        optionsContainer.addEventListener('click', (e) => {
          const optionDiv = e.target.closest('.option-btn');
          if (!optionDiv) return;
          const optionValue = optionDiv.getAttribute('data-option-value');
          if (optionValue) {
            selectOption(qIndex, optionValue);
          }
        });
      }

      const prevBtn = document.getElementById('prevBtn');
      if (prevBtn) {
        prevBtn.addEventListener('click', goToPrev);
      }

      const nextBtn = document.getElementById('nextBtn');
      if (nextBtn) {
        nextBtn.addEventListener('click', goToNext);
      }

      const finishBtn = document.getElementById('finishBtn');
      if (finishBtn) {
        finishBtn.addEventListener('click', finishTest);
      }
    }

    // 结果界面
    function renderResultScreen() {
      const type = state.mbtiResult;
      const description = typeDescriptions[type] || "独特的性格组合，兼具多种特质。";

      const html = `
        <h1>🌿 你的MBTI类型</h1>
        <div class="subhead">性格解码</div>
        
        <div class="result-box">
          <div class="mbti-type">${type}</div>
          <div class="type-desc">${description}</div>
          <div style="margin-top:1.2rem; color:#6a5544; font-size:0.95rem;">
            基于你对 ${questions.length} 个问题的回答
          </div>
          <button class="btn primary restart-btn" id="restartButton">🔄 重新测试</button>
        </div>
        <div style="text-align:center; margin-top:1.2rem; color:#b2947c; font-size:0.8rem;">
          * 本测试仅供娱乐参考，非专业心理评估
        </div>
      `;

      appContainer.innerHTML = html;

      const restartBtn = document.getElementById('restartButton');
      if (restartBtn) {
        restartBtn.addEventListener('click', restartTest);
      }
    }

    // 初始化渲染
    renderApp();
  })();
</script>
